1. VCC TO 5V 
2. GND TO GND
3. DO TO PIN 3
#define LDR 3      
#define LED 13     

bool headlightsOn = false;

void setup() {
  Serial.begin(9600);
  pinMode(LDR, INPUT);
  pinMode(LED, OUTPUT);
}

void loop() {

  int lightState = digitalRead(LDR);

  Serial.print("LDR State: ");
  Serial.println(lightState);

  // 🌙 DARK → BOTH LEDs ON
  if (!headlightsOn && lightState == 1) {
    headlightsOn = true;
    digitalWrite(LED, HIGH);
    Serial.println("DARK → HEADLIGHTS ON");
  }

  // ☀️ BRIGHT → BOTH LEDs OFF
  else if (headlightsOn && lightState == 0) {
    headlightsOn = false;
    digitalWrite(LED, LOW);
    Serial.println("BRIGHT → HEADLIGHTS OFF");
  }

  delay(200);
}
