1. VCC TO 5V
2. GND TO GND
3. AO TO A1
4. DO TO LEAVE IT
5. LED PIN 8

#define waterSensor A1
#define ledPin 8

int value = 0;
int threshold = 500;  // You will tune this

void setup() {
  Serial.begin(9600);
  pinMode(ledPin, OUTPUT);
}

void loop() {
  value = analogRead(waterSensor);
  
  Serial.print("Water Sensor Value: ");
  Serial.println(value);

  if (value < threshold) {
    digitalWrite(ledPin, HIGH);  // LED ON
  } else {
    digitalWrite(ledPin, LOW);   // LED OFF
  }

  delay(200);
}
